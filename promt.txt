```python
   # Strategy: Token bucket
   
   def check_rate_limit(user_id: str, endpoint: str):
       key = f"ratelimit:{user_id}:{endpoint}"
       current = redis.incr(key)
       if current == 1:
           redis.expire(key, 60)
       return current <= RATE_LIMIT
```
